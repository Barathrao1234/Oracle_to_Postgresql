WITH batch_totals AS (
    SELECT
        b.batch_id,
        b.acct_id,
        b.src_ccy_tx,
        FN_IP_CNT(b.batch_id)      AS in_progress_cnt,
        FN_PEND_CNT(b.batch_id)    AS pending_cnt
    FROM alt_batch b
    WHERE b.batch_status = 'OPEN'
)
SELECT *
FROM (
    SELECT
        m.acct_id,
        m.src_ccy_tx,
        m.auth_hist         AS auth_hist_coll,
        m.totals_coll        AS totals_coll,
        FN_REJ_PART_IND(m.acct_id, m.src_ccy_tx)   AS rej_part_ind,
        FN_POST_PART_IND(m.acct_id, m.src_ccy_tx)  AS post_part_ind,
        FN_CHG_IND(m.acct_id)                      AS chg_ind,
        t.in_progress_cnt,
        t.pending_cnt
    FROM merchant_txn m
    JOIN batch_totals t
        ON m.acct_id = t.acct_id
       AND m.src_ccy_tx = NVL(t.src_ccy_tx, m.src_ccy_tx)
    <<JOIN_ALT_BATCH_OUT>>
    WHERE m.status_cd = 'A'
      AND EXISTS (
            SELECT 1
            FROM TABLE(m.auth_hist) ah
            WHERE ah.auth_state IN ('APPROVED', 'PENDING')
          )
      AND CAST(m.state_coll AS ALT_STATE_COLL_T) IS NOT NULL
      AND m.rq_coll IS NOT EMPTY
    ORDER BY m.acct_id, m.src_ccy_tx
)
WHERE ROWNUM <= :limitRows;
